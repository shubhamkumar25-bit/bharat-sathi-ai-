rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ================================
    // USER PROFILE
    // ================================
    match /users/{userId} {

      // User can read only their own profile
      allow read: if request.auth != null
                  && request.auth.uid == userId;

      // User can write only their own profile
      // Prevent user from assigning admin/super_admin role
      allow write: if request.auth != null
                    && request.auth.uid == userId
                    && (
                      // New document without role
                      !request.resource.data.keys().hasAny(['role'])

                      // User can explicitly have normal user role
                      || request.resource.data.role == 'user'

                      // Existing role cannot be changed by the user
                      || (
                        resource != null
                        && resource.data.role == request.resource.data.role
                      )
                    );

      // ================================
      // USER SUBCOLLECTIONS
      // ================================
      // Examples:
      // /users/{userId}/chatMessages/{messageId}
      // /users/{userId}/resumes/{resumeId}
      // /users/{userId}/bookmarks/{bookmarkId}
      // /users/{userId}/conversations/{conversationId}
      // /users/{userId}/activity/{activityId}

      match /{allSubcollections=**} {
        allow read, write: if request.auth != null
                            && request.auth.uid == userId;
      }
    }

    // ================================
    // DEFAULT DENY
    // ================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
