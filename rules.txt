rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    // ================================
    // USER PROFILES & SUBCOLLECTIONS
    // ================================
    match /users/{userId} {
      // User can read their own profile, or admins can read all profiles
      allow read: if request.auth != null;

      // User can create & update their profile
      allow write: if request.auth != null && request.auth.uid == userId;

      // User subcollections (chatMessages, resumes, bookmarks, etc.)
      match /{allSubcollections=**} {
        allow read, write: if request.auth != null && request.auth.uid == userId;
      }
    }

    // ================================
    // ANALYTICS & LOG EVENTS
    // ================================
    match /loginEvents/{eventId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
    }

    match /pageViews/{viewId} {
      allow create: if true;
      allow read: if request.auth != null;
    }

    match /featureEvents/{eventId} {
      allow create: if request.auth != null;
      allow read: if request.auth != null;
    }

    // ================================
    // DEFAULT DENY
    // ================================
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
